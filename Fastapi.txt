from pydantic import field_validator

class Student(BaseModel):
	id     : str = Field(... , examples=["S001"])
	name   : str = Field(..., min_length=3 , max_length=50)
	age    : int = Field(... , gt=0 , le=100)
	course : str = Field(... , max_length=50)
	marks  : int = Field(... , ge=0 , le=100)
	email  : EmailStr

	
	@field_validator("name") kis field pe lagana ha 
	
	@classmethod
	def check_name(cls , value): cls refers to class Student , and value is name
		
		if not value.replace(" " , "").isalpha():
			raise ValueError(" Name must contain only letters and spaces ")
		return value.title()   (title return suprabh sharma -> Suprabh Sharma)

NOTE -- We will not give grade pydantic will automatically calculate grade in terms of marks

	@property ( this decorator comes along with property )
	def grade(self) -> str:  ### here self refers to the object and how to access it self.field_name like self.id or self.name or self.age or self.marks
		if self.marks >= 90 : 
			return "A"
		elif self.marks >= 80 : 
			return "B"
		else :
			return "C"

@app.delete('/delete/{student_id}')
def delete_student(student_id : str = Path(... , description = "The Id of the Student That you want to delete" , examples =["S001"]):
	data = load_data()
	if student_id not in data:
		raise HTTPException( status_code = 404 , detail = f"Student with id {student_id} not found")

	del data[student_id]
	save_data(data)

	return { "message" : "The Student has been deleted successfully"}
